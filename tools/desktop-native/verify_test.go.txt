func TestNativeAcceptanceOverlayGate(t *testing.T) {
	path := filepath.Join(nativeIOAcceptanceDirectory, "native-controlled.flv")
	data, err := os.ReadFile(path)
	if err != nil { t.Fatal(err) }
	full, err := flv.Scan(bytes.NewReader(data), uint64(len(data)), flv.Options{})
	if err != nil || !full.Complete || full.HasErrors() || full.Index().Len() != 32 {
		t.Fatalf("generated structural fixture: %+v, %v", full, err)
	}
	s, err := Open(context.Background(), path, flv.Options{})
	if err != nil { t.Fatal(err) }
	t.Cleanup(func() {
		// On failure unblock only this test's gate. Keep all receipts as evidence.
		if _, err := os.Stat(filepath.Join(nativeIOAcceptanceDirectory, "release.json")); errors.Is(err, os.ErrNotExist) {
			_ = nativeIOAcceptanceWrite("release.json", nativeIOAcceptanceReceipt{})
		}
		ctx, cancel := context.WithTimeout(context.Background(), 2*time.Second)
		defer cancel()
		_ = s.Close(ctx)
	})
	if nativeIOAcceptanceMode != "Scan" {
		verifyNativeContentClose(t, s)
		return
	}
	deadline := time.Now().Add(3*time.Second)
	for {
		if _, err := os.Stat(filepath.Join(nativeIOAcceptanceDirectory, "entered.json")); err == nil { break }
		if time.Now().After(deadline) { t.Fatal("source did not reach gate") }
		time.Sleep(time.Millisecond)
	}
    if got := s.Snapshot(); got.State != session.StateRunning { t.Fatalf("at gate: %+v", got) }
	t.Log("before cancellation: running, gated before read 23 at offset 163")
	s.Cancel()
    if got := s.Snapshot(); got.State != session.StateCanceling { t.Fatalf("before release: %+v", got) }
	t.Log("before release: canceling")
	if err := nativeIOAcceptanceWrite("release.json", nativeIOAcceptanceReceipt{}); err != nil { t.Fatal(err) }
	ctx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
	defer cancel()
	if err := s.Wait(ctx); !errors.Is(err, context.Canceled) { t.Fatalf("wait: %v", err) }
	final := s.Snapshot()
    if final.State != session.StateCanceled || final.Progress.TagCount != 10 || final.Complete || final.Closed {
		t.Fatalf("canceled prefix: %+v", final)
	}
	page, err := s.Page(0, 100)
	if err != nil || len(page.Records) != 10 { t.Fatalf("partial page: %+v %v", page, err) }
	for i, tag := range page.Records {
		if tag.ID != uint64(i) || tag.Offset != uint64(13+15*i) { t.Fatalf("partial tag %d: %+v", i, tag) }
	}
    if _, err := os.Stat(filepath.Join(nativeIOAcceptanceDirectory, "closed.json")); !errors.Is(err, os.ErrNotExist) { t.Fatalf("scan cancellation closed the retained source: %v", err) }
    t.Logf("after release: canceled, retained IDs 0-9, source remains owned: %+v", final)
    if _, err := s.StartDetail(1, 0); err != nil { t.Fatalf("detail of committed prefix: %v", err) }
    for {
        detail, err := s.GetDetail(1)
        if err != nil { t.Fatal(err) }
        if detail.State != "pending" {
            if detail.Err != nil || detail.Detail == nil || detail.Detail.Kind != "amf0ScriptData" { t.Fatalf("prefix detail: %+v", detail) }
            break
        }
        if ctx.Err() != nil { t.Fatal("prefix detail did not finish") }
        time.Sleep(time.Millisecond)
    }
    if selected, err := s.StartSummary(2); err != nil || selected.State != "pending" { t.Fatalf("prefix Summary start: %+v %v", selected, err) }
    for {
        summary, err := s.GetSummary(2)
        if err != nil { t.Fatal(err) }
        if summary.State != "pending" {
            if summary.Err != nil || summary.State != "ready" || summary.Processed != 10 || summary.Total != 10 || summary.Overview == nil || summary.Overview.ScanComplete || summary.Overview.IndexedTags != "10" { t.Fatalf("prefix Summary: %+v", summary) }
            break
        }
        if ctx.Err() != nil { t.Fatal("prefix Summary did not finish") }
        time.Sleep(time.Millisecond)
    }
    t.Log("canceled scan prefix: Summary completed all 10 retained Tags with scanComplete=false")
    if err := s.Close(ctx); !errors.Is(err, context.Canceled) { t.Fatalf("close: %v", err) }
    if _, err := os.Stat(filepath.Join(nativeIOAcceptanceDirectory, "closed.json")); err != nil { t.Fatal(err) }
	if !s.Snapshot().Closed { t.Fatal("session Close was not recorded") }
    if _, err := s.Page(0, 100); !errors.Is(err, session.ErrClosed) { t.Fatalf("page after close: %v", err) }
}

func verifyNativeContentClose(t *testing.T, s *Document) {
	ctx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
	defer cancel()
	if err := s.Wait(ctx); err != nil { t.Fatalf("complete scan before content read: %v", err) }
	if got := s.Snapshot(); !got.State.Terminal() || !got.Complete || got.Closed || got.Progress.TagCount != 32 {
		t.Fatalf("scan before detail: %+v", got)
	}
	for _, name := range []string{"entered.json", "closed.json"} {
		if _, err := os.Stat(filepath.Join(nativeIOAcceptanceDirectory, name)); !errors.Is(err, os.ErrNotExist) {
			t.Fatalf("scan unexpectedly produced %s: %v", name, err)
		}
	}
	if nativeIOAcceptanceMode == "Summary" {
		selected, err := s.StartSummary(1)
		if err != nil || selected.State != "pending" { t.Fatalf("Summary start: %+v %v", selected, err) }
	} else if nativeIOAcceptanceMode == "Hex" {
		selected, err := s.StartHex(1, nativeIOAcceptanceOffset, 512)
		if err != nil || selected.State != "pending" { t.Fatalf("Hex start: %+v %v", selected, err) }
	} else {
		selected, err := s.StartDetail(1, 0)
		if err != nil || selected.State != "pending" { t.Fatalf("detail start: %+v %v", selected, err) }
	}
	for {
		if _, err := os.Stat(filepath.Join(nativeIOAcceptanceDirectory, "entered.json")); err == nil { break }
		if ctx.Err() != nil { t.Fatal("content read did not reach gate") }
		time.Sleep(time.Millisecond)
	}
	t.Logf("scan complete: 32 Tags, source retained; %s gated before read %d at offset %d", nativeIOAcceptanceMode, nativeIOAcceptanceReadCall, nativeIOAcceptanceOffset)
	if err := nativeIOAcceptanceWrite("close-command.json", nativeIOAcceptanceReceipt{}); err != nil { t.Fatal(err) }
	poll, stop := context.WithCancel(context.Background())
	stop()
	if err := s.Close(poll); !errors.Is(err, context.Canceled) { t.Fatalf("blocked Close must stop waiting: %v", err) }
	if got := s.Snapshot(); !got.Closed || got.State.Terminal() || got.Progress.TagCount != 32 {
		t.Fatalf("blocked close advertised a reusable source: %+v", got)
	}
	if _, err := os.Stat(filepath.Join(nativeIOAcceptanceDirectory, "closed.json")); !errors.Is(err, os.ErrNotExist) { t.Fatalf("closed while content read was active: %v", err) }
	if _, err := s.StartDetail(2, 0); !errors.Is(err, session.ErrClosed) { t.Fatalf("new detail while closing: %v", err) }
	if _, err := s.GetDetail(1); !errors.Is(err, session.ErrClosed) { t.Fatalf("old detail while closing: %v", err) }
	if _, err := s.StartHex(3, 13, 512); !errors.Is(err, session.ErrClosed) { t.Fatalf("new Hex while closing: %v", err) }
	if _, err := s.GetHex(1); !errors.Is(err, session.ErrClosed) { t.Fatalf("old Hex while closing: %v", err) }
	if _, err := s.StartSummary(4); !errors.Is(err, session.ErrClosed) { t.Fatalf("new Summary while closing: %v", err) }
	if _, err := s.GetSummary(1); !errors.Is(err, session.ErrClosed) { t.Fatalf("old Summary while closing: %v", err) }
	if _, err := s.CancelSummary(1); !errors.Is(err, session.ErrClosed) { t.Fatalf("cancel Summary while closing: %v", err) }
	if _, err := s.SummaryEvents(1, "aac", 0, 100); !errors.Is(err, session.ErrClosed) { t.Fatalf("Summary events while closing: %v", err) }
	if _, err := s.Page(0, 100); !errors.Is(err, session.ErrClosed) { t.Fatalf("page while closing: %v", err) }
	t.Log("close before release: closed=true, terminal=false, source retained, details, Hex, Summary and pages rejected")
	if err := nativeIOAcceptanceWrite("release.json", nativeIOAcceptanceReceipt{}); err != nil { t.Fatal(err) }
	if err := s.Close(ctx); err != nil { t.Fatalf("Close after content release: %v", err) }
	if got := s.Snapshot(); !got.Closed || !got.State.Terminal() { t.Fatalf("Close never released source: %+v", got) }
	if _, err := os.Stat(filepath.Join(nativeIOAcceptanceDirectory, "closed.json")); err != nil { t.Fatal(err) }
	t.Log("release completed the active read; document Close returned nil and source closed")
}
